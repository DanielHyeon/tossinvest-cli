def run_tests(pkg, pat):
    try:
        r = sh(["go", "test", pkg, "-count=1", "-timeout", "120s", "-run", pat], cwd=COPY, timeout=400)
    except subprocess.TimeoutExpired:
        return "TIMEOUT", ""
    out = r.stdout + r.stderr
    if r.returncode == 0:
        return "PASS", out
    if "[build failed]" in out or "\n#" in out[:2000] and "FAIL\t" not in out:
        return "BUILD_FAIL", out
    if "panic: test timed out" in out:
        return "TIMEOUT", out
    return "FAIL", out
ledger = []
for pkg, pat in (ENGINE, CMD):
    st, out = run_tests(pkg, pat)
    ledger.append({"id": "CONTROL", "pkg": pkg, "result": st})
    if st != "PASS":
        print(out[-3000:]); sys.exit(f"unmutated control is not GREEN for {pkg}: {st}")
for mid, what, path, old, new, (pkg, pat) in M:
    full = os.path.join(COPY, path)
    text = open(full).read()
    assert text.count(old) == 1, f"{mid}: anchor count {text.count(old)}"
    text = text.replace(old, new)
    if mid in EXTRA:
        p2, o2, n2 = EXTRA[mid]
        assert p2 == path and text.count(o2) == 1, f"{mid}: extra anchor"
        text = text.replace(o2, n2)
    open(full, "w").write(text)
    diff = sh(["git", "-C", COPY, "diff", "--stat"]).stdout.strip()
    assert diff, f"{mid}: mutation not applied"
    t0 = time.time(); st, out = run_tests(pkg, pat); dt = time.time() - t0
    verdict = {"FAIL": "CAUGHT", "TIMEOUT": "CAUGHT(timeout)", "PASS": "SURVIVED", "BUILD_FAIL": "BUILD_FAIL"}[st]
    failing = sorted({l.split()[2] for l in out.splitlines() if l.startswith("--- FAIL:")})
    ledger.append({"id": mid, "what": what, "file": path, "pkg": pkg, "result": verdict, "failing": failing, "seconds": round(dt, 1)})
    print(mid, verdict, failing[:4], flush=True)
    must(sh(["git", "-C", COPY, "checkout", "--", path]))
    assert sh(["git", "-C", COPY, "status", "--porcelain"]).stdout.strip() == "", f"{mid}: copy not clean after revert"
