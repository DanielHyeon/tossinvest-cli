465		if o.opts.SLO != nil && o.opts.SLO.FillDetectionBehind() {
466			// Yield the whole cycle, and let the outage clock keep running. See the
467			// file header: an unobserved position is unprotected whether the reason
468			// was a failure or a choice.
469			cycle.Deferred = true
470			o.checkOutage(ctx, &cycle)
471			return cycle
472		}
473	
474		states, err := o.workingSet(ctx, &cycle)
475		if err != nil {
476			cycle.Err = err
477			return cycle

479		if len(states) == 0 {
480			// a090: 보유·대상 표시가 있는데 전부 탈락했으면 그 주기도 미관측으로 셈 — 표시가 없으면 기록을 비움.
481			o.settleUnobserved(ctx, &cycle)

483			// unprotected. The outage clock is reset rather than left running: an
484			// account with no positions that has not read a price is not in an
485			// outage, and escalating it would block entries for a condition that
486			// only entering could clear.
487			o.lastObserved = o.clk.Now()
488			o.outageRaised = false

490		}
491		quotes, err := o.observe(ctx, symbolsOf(states), observation)
492		if err != nil {
493			cycle.Err = err
494			o.checkOutage(ctx, &cycle)
495			return cycle
496		}
497		o.lastObserved = o.clk.Now()
498		o.outageRaised = false
499		cycle.Observed = len(quotes)

501		for _, state := range states {

508				o.noteUnobservedCause(&cycle, state.position.ID, unobservedNoQuote)

510			}

513				// judgement. Keep its state untouched until the next one-batch cycle.
514				o.noteUnobservedCause(&cycle, state.position.ID, unobservedQuoteExpired)
515				continue

842	func (o *ExitObserver) observe(ctx context.Context, symbols []string, observation cycleObservation) (map[string]observedQuote, error) {
843		var quotes []domain.Quote
844		var out map[string]observedQuote
845		needsFallback := false
846		err := o.opts.Retrier.Query(ctx, execgw.QueryPrice, func(ctx context.Context) error {
847			var qerr error
848			quotes, qerr = o.opts.Prices.Prices(ctx, symbols)
849			if qerr != nil {
850				return qerr
851			}
852			// Keep this process-local anchor separate from the persisted UTC evidence
853			// time. clock.LeaseAnchor preserves the system clock's monotonic reading,
854			// while injected clocks deterministically use their Now/Since pair.
855			leaseAnchor := clock.LeaseAnchor(o.clk)
856			validatedAt := leaseAnchor.UTC()
857			wanted := make(map[string]struct{}, len(symbols))
858			for _, symbol := range symbols {
859				wanted[strings.ToUpper(strings.TrimSpace(symbol))] = struct{}{}
860			}
861			next := make(map[string]observedQuote, len(quotes))
862			for _, q := range quotes {
863				symbol := strings.ToUpper(strings.TrimSpace(q.Symbol))
864				if _, needed := wanted[symbol]; !needed || q.Last <= 0 || math.IsNaN(q.Last) || math.IsInf(q.Last, 0) {
865					continue
866				}
867				effectiveAt, source := q.FetchedAt.UTC(), "quote_fetched_at"
868				if q.FetchedAt.IsZero() {
869					effectiveAt, source = validatedAt, ""
870					needsFallback = true
871				} else if effectiveAt.After(validatedAt) ||
872					validatedAt.Sub(effectiveAt) > execgw.QueryPriceEvidenceDuration {
873					continue
874				}
875				next[symbol] = observedQuote{Price: decimalOf(q.Last), FetchedAt: effectiveAt,
876					Source: source, leaseAnchor: leaseAnchor}
877			}
878			if len(next) == 0 {
879				return execgw.NewEvidenceInvalidError(fmt.Sprintf("no valid quote evidence for %d managed symbol(s)", len(symbols)))
880			}
881			out = next
882			return nil
883		})
884		if err != nil {
885			return nil, err
886		}
887		if needsFallback {
888			// Official fetched-at evidence never pays for this scan. Only a successful
889			// managed quote that actually needs the cycle fallback recovers the durable
890			// maximum, and it does so after the one broker read so a local journal error
891			// cannot make Retrier repeat that read.
892			maximum, err := o.opts.Journal.MaxExitObservationCycle(ctx, o.opts.AccountRef)
893			if err != nil {
894				return nil, err
895			}
896			for {
897				current := o.observationSequence.Load()
898				if current >= maximum || o.observationSequence.CompareAndSwap(current, maximum) {
899					break
900				}
901			}
902			sequence := o.observationSequence.Add(1)
903			source := "cycle:" + strconv.FormatUint(sequence, 10)
904			for symbol, quote := range out {
905				if quote.Source != "" {
906					continue
907				}
908				quote.Source = source
909				out[symbol] = quote
910			}
911		}
912		return out, nil
913	}
