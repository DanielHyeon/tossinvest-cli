package main

import (
	"encoding/json"
	"fmt"
	"io"
	"os"
)

func main() {
	b, err := io.ReadAll(os.Stdin)
	if err != nil {
		panic(err)
	}
	out, err := json.Marshal(string(b))
	if err != nil {
		panic(err)
	}
	fmt.Print(string(out))
}
