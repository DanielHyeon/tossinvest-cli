# 옛(pre-edit-ast)/새(function-logic/*/ast.json) 분기를 (종류, 그 줄의 소스 문장)으로 difflib 정렬해 재번호 표를 씀.
# 사용: python3 analysis/renumber.py.txt <old.ast.json> <new.ast.json> <base-rev>   (저장소 루트에서)
import difflib, json, subprocess, sys
old, new, base = json.load(open(sys.argv[1])), json.load(open(sys.argv[2])), sys.argv[3]
src_old = subprocess.run(["git", "show", f"{base}:{old['file']}"], capture_output=True, text=True, check=True).stdout.splitlines()
src_new = open(new["file"]).read().splitlines()
def seq(d, src): return [(b["kind"], src[b["at"]["line"] - 1].strip()) for b in d.get("branches") or []]
a, b = seq(old, src_old), seq(new, src_new)
ob, nb = old.get("branches") or [], new.get("branches") or []
print(f"# {new['function']}: 옛 분기 {len(ob)} → 새 분기 {len(nb)} (base {base})")
for tag, i1, i2, j1, j2 in difflib.SequenceMatcher(a=a, b=b, autojunk=False).get_opcodes():
    if tag == "equal":
        for k in range(i2 - i1):
            o, n = ob[i1 + k], nb[j1 + k]
            print(f"same    {o['id']}({o['kind']}@{o['at']['line']}) -> {n['id']}({n['kind']}@{n['at']['line']})  {a[i1+k][1][:70]}")
    else:
        for k in range(i1, i2):
            print(f"{'removed' if tag=='delete' else 'changed'} {ob[k]['id']}({ob[k]['kind']}@{ob[k]['at']['line']})  {a[k][1][:70]}")
        for k in range(j1, j2):
            print(f"{'added  ' if tag=='insert' else 'changed'} -> {nb[k]['id']}({nb[k]['kind']}@{nb[k]['at']['line']})  {b[k][1][:70]}")
