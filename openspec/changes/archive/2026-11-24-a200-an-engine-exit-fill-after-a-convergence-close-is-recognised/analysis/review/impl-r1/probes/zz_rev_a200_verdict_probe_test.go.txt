package journal

import "testing"

// 리뷰 탐침(커밋 대상 아님): (라) 경계 · 이상 값.
func TestRevA200ProbeVerdictEdges(t *testing.T) {
	for _, c := range []struct{ prev, newq, intent string }{
		{"31", "0", "31.000000001"}, {"31", "0", "30.999999999"}, {"31", "0", "31.0"}, {"31", "0", " 31 "},
		{"31", "0", "-1"}, {"31", "0", "0"}, {"31", "0", "1e1"}, {"31", "0", "abc"}, {"31", "0", "+31"},
		{"0.5", "0", "0.5"}, {"-5", "0", "1"}, {"31", "-0", "31"}, {"31", "0.000", "31"},
	} {
		f := a200GoodFacts()
		f.Adjustments[0].PrevQuantity, f.Adjustments[0].NewQuantity, f.IntentQuantity = c.prev, c.newq, c.intent
		t.Logf("REVEDGE|prev=%q new=%q intent=%q -> %v", c.prev, c.newq, c.intent, recognitionVerdict(f))
	}
}
